/**
 * Register a new user
 * @name POST/users
 * @function
 * @memberof module:server
 * @param {Object} req.body - User registration data
 * @param {string} req.body.username - Username (min 5 characters, alphanumeric)
 * @param {string} req.body.password - User password
 * @param {string} req.body.email - Valid email address
 * @param {Date} [req.body.birthday] - User's birthday (optional)
 * @returns {Object} 201 - Created user object
 * @returns {Object} 400 - User already exists
 * @returns {Object} 422 - Validation errors
 * @returns {Object} 500 - Server error
 * @example
 * // Request body
 * {
 *   "username": "john_doe",
 *   "password": "securePass123",
 *   "email": "john@example.com",
 *   "birthday": "1990-01-15"
 * }
 */
app.post('/users', 
  [
    check('username', 'Username is required').isLength({ min: 5 }),
    check('username', 'Only alphanumeric values allowed').isAlphanumeric(),
    check('password', 'Password is required').not().isEmpty(),
    check('email', 'Invalid email address').isEmail()
  ], async (req, res) => {
    let errors = validationResult(req);

    if(!errors.isEmpty()) {
      return res.status(422).json({ errors: errors.array() });
    }

  let hashedPassword = Users.hashPassword(req.body.password);

  await Users.findOne({ username: req.body.username })
    .then((user) => {
      if (user) {
        return res.status(400).send(`${req.body.username} already exists`);
      }
      else {
        Users.create({
          username: req.body.username,
          password: hashedPassword,
          email: req.body.email,
          birthday: req.body.birthday
        })
          .then((user) => {
            res.status(201).json(user)
          })
          .catch((error) => {
            console.error(error);
            res.status(500).send(`Error: ${error}`);
          });
      }
    })
    .catch((error) => {
      console.error(error);
      res.status(500).send(`Error: ${error}`);
    });
});

/**
 * Add a movie to user's favorites list
 * @name POST/users/:username/favoriteMovies/:movieId
 * @function
 * @memberof module:server
 * @param {string} req.params.username - Username
 * @param {string} req.params.movieId - MongoDB ObjectId of the movie
 * @returns {Object} 200 - Updated user object
 * @returns {Object} 500 - Server error
 * @security JWT
 * @example
 * // POST /users/john_doe/favoriteMovies/507f1f77bcf86cd799439011
 */
app.post('/users/:username/favoriteMovies/:movieId', passport.authenticate('jwt', { session: false }), async (req, res) => {
  await Users.findOneAndUpdate({ username: req.params.username }, {
    $push: {
      favoriteMovies: req.params.movieId
    },
  },
    { new: true })
    .then((updatedUser) => {
      res.json(updatedUser);
    })
    .catch((err) => {
      console.error(err);
      res.status(500).send(`Error: ${err}`);
    });
});

/**
 * Get director information by name
 * @name GET/movies/directors/:name
 * @function
 * @memberof module:server
 * @param {string} req.params.name - Director's name
 * @returns {Object} 200 - Director object with name, biography, birthYear, deathYear
 * @returns {Object} 404 - Director not found
 * @returns {Object} 500 - Server error
 * @security JWT
 * @description Searches for a director within movie documents and returns their details
 * @example
 * // GET /movies/directors/Francis%20Ford%20Coppola
 */
app.get('/movies/directors/:name', passport.authenticate('jwt', { session: false }), async (req, res) => {
  try {
    const directorName = req.params.name;

    // Search for a movie that contains the director
    const movie = await Movies.findOne({
      'directors.name': new RegExp(`^${directorName}$`, 'i')
    });

    if (!movie) {
      return res.status(404).send('Director not found in any movie.');
    }

    // Find the matching director object from the array
    const director = movie.directors.find(d =>
      d.name.toLowerCase() === directorName.toLowerCase()
    );

    if (!director) {
      return res.status(404).send('Director found in movie, but could not extract full details.');
    }

    res.status(200).json({
      name: director.name,
      biography: director.biography,
      birthYear: director.birthYear,
      deathYear: director.deathYear
    });

  } catch (err) {
    console.error(err);
    res.status(500).send(`Error: ${err.message}`);
  }
});

/**
 * Update user information
 * @name PUT/users/:username
 * @function
 * @memberof module:server
 * @param {string} req.params.username - Username to update
 * @param {Object} req.body - Updated user data
 * @param {string} req.body.username - New username (min 5 characters, alphanumeric)
 * @param {string} req.body.password - New password
 * @param {string} req.body.email - New email address
 * @param {Date} [req.body.birthday] - New birthday (optional)
 * @returns {Object} 200 - Updated user object
 * @returns {Object} 400 - Permission denied (user can only update their own account)
 * @returns {Object} 422 - Validation errors
 * @returns {Object} 500 - Server error
 * @security JWT
 * @description Users can only update their own account information
 * @example
 * // PUT /users/john_doe
 * // Request body
 * {
 *   "username": "john_doe_new",
 *   "password": "newPassword123",
 *   "email": "newemail@example.com",
 *   "birthday": "1990-01-15"
 * }
 */
app.put('/users/:username', passport.authenticate('jwt', { session: false }), 
[
  check('username', 'Username is required').isLength({ min: 5 }),
  check('username', 'Only alphanumeric values allowed').isAlphanumeric(),
  check('password', 'Password is required').not().isEmpty(),
  check('email', 'Invalid email address').isEmail()
], async (req, res) => {
  let errors = validationResult(req);

  if(!errors.isEmpty()) {
    return res.status(422).json({ errors: errors.array() });
  }

  if (req.user.username !== req.params.username) {
    return res.status(400).send('Permission denied');
  }
  
  await Users.findOneAndUpdate({ username: req.params.username }, {
    $set: {
      username: req.body.username,
      password: req.body.password,
      email: req.body.email,
      birthday: req.body.birthday
    }
  },
    { new: true })
    .then((updatedUser) => {
      res.json(updatedUser);
    })
    .catch((err) => {
      console.error(err);
      res.status(500).send(`Error: ${err}`);
    });
});

/**
 * Delete a user account
 * @name DELETE/users/:username
 * @function
 * @memberof module:server
 * @param {string} req.params.username - Username to delete
 * @returns {string} 200 - Success message confirming deletion
 * @returns {string} 400 - User not found
 * @returns {Object} 500 - Server error
 * @security JWT
 * @description Permanently removes a user account from the database
 * @example
 * // DELETE /users/john_doe

app.delete('/users/:username', passport.authenticate('jwt', { session: false }), async (req, res) => {

    .then((user) => {

        res.status(200).send(`${req.params.username} was deleted`);
      }
    })
    .catch((err) => {
      console.error(err);
      res.status(500).send(`Error: ${err}`);
    });
});

/**
 * Remove a movie from user's favorites list
 * @name DELETE/users/:username/favoriteMovies/:movieId
 * @function
 * @memberof module:server
 * @param {string} req.params.username - Username
 * @param {string} req.params.movieId - MongoDB ObjectId of the movie to remove
 * @returns {Object} 200 - Success message and updated user object
 * @returns {string} 400 - Invalid movie ID format
 * @returns {string} 404 - User not found
 * @returns {Object} 500 - Server error
 * @security JWT
 * @description Removes a movie from the user's favorite movies array
 * @example
 * // DELETE /users/john_doe/favoriteMovies/507f1f77bcf86cd799439011
 * // Response: { "message": "Movie 507f1f77bcf86cd799439011 removed", "user": {...} }
 */
app.delete('/users/:username/favoriteMovies/:movieId', passport.authenticate('jwt', { session: false }), async (req, res) => {
  try {
    const { username, movieId } = req.params;

    // Validate movieId as a valid MongoDB ObjectId
    if (!mongoose.Types.ObjectId.isValid(movieId)) {
      return res.status(400).send(`Invalid movie ID: ${ movieId }`);
    }

    const updatedUser = await Users.findOneAndUpdate(
      { username },
      { $pull: { favoriteMovies: movieId } },
      { new: true }
    );

    if (!updatedUser) {
      return res.status(404).send(`${username} not found`);
    }

    res.status(200).json({
      message: `Movie ${movieId} removed`,
      user: updatedUser
    });
  } catch (err) {
    console.error(err);
    res.status(500).send(`Server error: ${ err.message }`);
  }
});
