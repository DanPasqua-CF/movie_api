/**
 * @file Mongoose schemas and models for the movie database application
 * @module models
 */

const mongoose = require('mongoose');
const bcrypt = require('bcrypt');

/**
 * Schema for director information
 * @typedef {Object} DirectorSchema
 * @property {string} name - Director's full name
 * @property {string} biography - Brief biography of the director
 * @property {Date} birthYear - Director's birth year
 * @property {Date} deathYear - Director's death year (if applicable)
 */
const directorSchema = new mongoose.Schema({
  name: String,
  biography: String,
  birthYear: Date,
  deathYear: Date
});

/**
 * Schema for movie genres
 * @typedef {Object} GenreSchema
 * @property {string} name - Genre name (required)
 * @property {string} description - Description of the genre
 */
const genreSchema = new mongoose.Schema({
  name: { type: String, required: true },
  description: String
});

/**
 * Schema for movies in the database
 * @typedef {Object} MovieSchema
 * @property {string} title - Movie title (required)
 * @property {mongoose.Schema.Types.ObjectId[]} genre - Array of genre IDs referencing Genre model
 * @property {string} description - Movie description (required)
 * @property {DirectorSchema[]} directors - Array of director objects
 * @property {string} imageUrl - URL to movie poster/image
 * @property {boolean} featured - Whether the movie is featured
 */
const movieSchema = new mongoose.Schema({
  title: { type: String, required: true },
  genre: [{ type: mongoose.Schema.Types.ObjectId, ref: "Genre" }],
  description: { type: String, required: true },
  directors: [directorSchema],
  imageUrl: String,
  featured: Boolean
});

/**
 * Schema for user accounts
 * @typedef {Object} UserSchema
 * @property {string} username - User's unique username (required)
 * @property {string} password - Hashed password (required)
 * @property {string} email - User's email address (required)
 * @property {Date} birthday - User's date of birth
 * @property {mongoose.Schema.Types.ObjectId[]} favoriteMovies - Array of movie IDs referencing Movie model
 */
const userSchema = new mongoose.Schema({
  username: { type: String, required: true },
  password: { type: String, required: true },
  email: { type: String, required: true },
  birthday: Date,
  favoriteMovies: [{ type: mongoose.Schema.Types.ObjectId, ref: 'Movie' }]
});

/**
 * Hashes a plain text password using bcrypt
 * @static
 * @param {string} password - Plain text password to hash
 * @returns {string} Hashed password with salt rounds of 10
 * @example
 * const hashedPassword = User.hashPassword('mySecretPassword');
 */
userSchema.statics.hashPassword = (password) => {
  return bcrypt.hashSync(password, 10);
};

/**
 * Validates a plain text password against the user's hashed password
 * @method
 * @param {string} password - Plain text password to validate
 * @returns {boolean} True if password matches, false otherwise
 * @example
 * const user = await User.findOne({ username: 'john_doe' });
 * const isValid = user.validatePassword('myPassword');
 */
userSchema.methods.validatePassword = function(password) {
  return bcrypt.compareSync(password, this.password);
};

/**
 * Mongoose model for movies
 * @type {mongoose.Model}
 */
const Movie = mongoose.model('Movie', movieSchema);

/**
 * Mongoose model for users
 * @type {mongoose.Model}
 */
const User = mongoose.model('User', userSchema);

/**
 * Mongoose model for genres
 * @type {mongoose.Model}
 */
const Genre = mongoose.model('Genre', genreSchema);

module.exports = { Movie, User, Genre };
