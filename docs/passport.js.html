/**
 * @file Passport configuration for local and JWT authentication strategies
 * @module passport
 * @requires passport
 * @requires passport-local
 * @requires passport-jwt
 * @requires mongoose
 * @requires ./models
 */

const { default: mongoose } = require('mongoose');

const passport = require('passport'),
  LocalStrategy = require('passport-local').Strategy,
  Models = require('./models'),
  passportJWT = require('passport-jwt');

const Users = Models.User;
const JWTStrategy = passportJWT.Strategy;
const ExtractJWT = passportJWT.ExtractJwt;

/**
 * Configures Passport Local Strategy for username/password authentication
 * @description Authenticates users with username and password credentials.
 * Includes validation to prevent NoSQL injection attacks by checking username format.
 * @strategy LocalStrategy
 * @param {string} username - User's username (alphanumeric only)
 * @param {string} password - User's password
 * @callback {Function} callback - Passport callback function
 * @param {Error|null} callback.error - Error object if authentication fails
 * @param {Object|boolean} callback.user - User object if successful, false if failed
 * @param {Object} callback.info - Additional info object with error messages
 * @throws {Error} Database errors during user lookup
 * @example
 * // Usage in route
 * passport.authenticate('local', { session: false }, (err, user, info) => {
 *   if (err || !user) {
 *     return res.status(400).json({ message: info.message });
 *   }
 *   // Generate token and return
 * });
 */
passport.use(
  new LocalStrategy(
    {
      usernameField: 'username',
      passwordField: 'password'
    },
    async (username, password, callback) => {
      try {
        // Validate input format to prevent NoSQL injections
        if (typeof username !== 'string' || !/^[a-zA-Z0-9]+$/.test(username)) {
          return callback(null, false, { message: 'Invalid username format' });
        }

        const user = await Users.findOne({ username: username.trim() });

        if (!user) {
          console.log('Incorrect username');
          return callback(null, false, { message: 'Incorrect username' });
        }

        if (!user.validatePassword(password)) {
          console.log('Incorrect password');
          return callback(null, false, { message: 'Incorrect password' });
        }

        console.log('Login successful');
        return callback(null, user);
      }
      catch (error) {
        console.log(error);
        return callback(error);
      }
    }
  )
);

/**
 * Configures Passport JWT Strategy for token-based authentication
 * @description Authenticates users via JWT tokens extracted from the Authorization header.
 * Validates the token's user ID before attempting database lookup.
 * @strategy JWTStrategy
 * @param {Object} jwtPayload - Decoded JWT payload
 * @param {string} jwtPayload._id - User's MongoDB ObjectId from the token
 * @param {string} jwtPayload.username - User's username from the token
 * @callback {Function} callback - Passport callback function
 * @param {Error|null} callback.error - Error object if authentication fails
 * @param {Object|boolean} callback.user - User object if successful, false if failed
 * @throws {Error} Database errors during user lookup
 * @example
 * // Usage in protected route
 * app.get('/movies', 
 *   passport.authenticate('jwt', { session: false }), 
 *   (req, res) => {
 *     // req.user contains authenticated user
 *   }
 * );
 * 
 * // Request header format
 * // Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
 */
passport.use(
  new JWTStrategy(
    {
      jwtFromRequest: ExtractJWT.fromAuthHeaderAsBearerToken(),
      secretOrKey: 'your_jwt_secret'
    }, 
    async (jwtPayload, callback) => {
      try {
        if (!mongoose.Types.ObjectId.isValid(jwtPayload._id)) {
          return callback(null, false);
        }

        const user = await Users.findById(jwtPayload._id);
        if (user) {
          return callback(null, user);
        } 
        else {
          return callback(null, false);
        }
      } 
      catch (error) {
        return callback(error);
      }
    }
  )
);
