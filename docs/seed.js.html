/**
 * @file Database seeding script for populating MongoDB with initial data
 * @module seed
 * @requires dotenv
 * @requires fs
 * @requires path
 * @requires mongoose
 * @requires ./models
 * @description This script reads JSON files containing movies, genres, and users data
 * and populates the MongoDB database. It clears existing data before seeding to ensure
 * a clean state. Run this script with: node seed.js
 */

require('dotenv').config();
const fs = require('fs');
const path = require('path');
const mongoose = require('mongoose');
const Models = require('./models');
const Movies = Models.Movie;
const Users = Models.User;
const Genres = Models.Genre;

/**
 * MongoDB connection URI loaded from environment variables
 * @constant {string}
 */
const mongoUri = process.env.MONGODB_URI;

/**
 * Seeds the database with initial data from JSON files
 * @async
 * @function seedData
 * @returns {Promise<void>} Resolves when seeding is complete
 * @throws {Error} If MongoDB connection fails, file reading fails, or data insertion fails
 * @description Performs the following operations:
 * 1. Connects to MongoDB using the URI from environment variables
 * 2. Clears all existing movies, genres, and users from the database
 * 3. Reads data from movies.json, genres.json, and users.json
 * 4. Inserts the data into their respective collections
 * 5. Closes the database connection
 * @example
 * // Run the seeding script
 * node seed.js
 * 
 * // Expected console output:
 * // Connected to MongoDB
 * // Inserted 3 genres
 * // Inserted 10 movies
 * // Inserted 5 users
 * // Seeding complete!
 */
async function seedData() {
  try {
    // Connect to MongoDB
    await mongoose.connect(mongoUri, {
      useNewUrlParser: true,
      useUnifiedTopology: true,
    });
    console.log('Connected to MongoDB');

    /**
     * Clear existing collections before seeding
     * @description Removes all documents from Movies, Genres, and Users collections
     * to ensure a clean database state before inserting new data
     */
    await Movies.deleteMany({});
    await Genres.deleteMany({});
    await Users.deleteMany({});

    /**
     * Read and parse JSON data files
     * @description Loads movie, genre, and user data from JSON files in the project directory
     */
    const moviesData = JSON.parse(fs.readFileSync(path.join(__dirname, 'movies.json'), 'utf-8'));
    const genresData = JSON.parse(fs.readFileSync(path.join(__dirname, 'genres.json'), 'utf-8'));
    const usersData = JSON.parse(fs.readFileSync(path.join(__dirname, 'users.json'), 'utf-8'));

    // Insert genres into database
    await Genres.insertMany(genresData);
    console.log(`Inserted ${genresData.length} genres`);

    // Insert movies into database
    await Movies.insertMany(moviesData);
    console.log(`Inserted ${moviesData.length} movies`);

    // Insert users into database
    await Users.insertMany(usersData);
    console.log(`Inserted ${usersData.length} users`);

    console.log('Seeding complete!');
  } catch (error) {
    console.error('Seeding failed:', error);
  } finally {
    /**
     * Close MongoDB connection
     * @description Ensures the database connection is properly closed
     * regardless of whether seeding succeeded or failed
     */
    mongoose.connection.close();
  }
}

/**
 * Execute the seeding function
 * @description Immediately invokes seedData() when the script is run
 */
seedData();
