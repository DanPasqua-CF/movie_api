/**
 * @file Authentication module for user login and JWT token generation
 * @module auth
 */

const jwtSecret = 'your_jwt_secret';

const jwt = require('jsonwebtoken'),
passport = require('passport');

require('./passport');

/**
 * Generates a JSON Web Token for authenticated users
 * @param {Object} user - The user object to encode in the token
 * @param {string} user.username - The username of the authenticated user
 * @returns {string} A signed JWT token valid for 7 days
 * @example
 * const token = generateJWTToken({ username: 'john_doe', _id: '12345' });
 */
let generateJWTToken = (user) => {
  return jwt.sign(user, jwtSecret, {
    subject: user.username,
    expiresIn: '7d',
    algorithm: 'HS256'
  });
}

/**
 * Configures authentication routes for the application
 * @param {Object} router - Express router instance
 * @returns {void}
 * @description Sets up POST /login endpoint with passport local authentication
 */
module.exports = (router) => {
  /**
   * POST /login
   * Authenticates user credentials and returns a JWT token
   * @name LoginEndpoint
   * @route {POST} /login
   * @bodyparam {string} username - User's username
   * @bodyparam {string} password - User's password
   * @returns {Object} 200 - Success response with user object and JWT token
   * @returns {Object} 400 - Error response with message
   * @example
   * // Request body
   * { "username": "john_doe", "password": "secret123" }
   * 
   * // Success response
   * { 
   *   "user": { "username": "john_doe", "_id": "12345" },
   *   "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..."
   * }
   */
  router.post('/login', (req, res) => {
    console.log('Received login request:', req.body);

    passport.authenticate('local', { session: false}, (error, user, info) => {
      if (error || !user) {
        console.log('Auth info', info);

        return res.status(400).json({
          message: info ? info.message : `Something isn't right`,
          user: user
        });
      }
      req.login(user, { session: false }, (error) => {
        if (error) {
          res.send(error);
        }

        let token = generateJWTToken(user.toJSON());
        return res.json({ user, token });
      });
    })(req, res);
  });
}
